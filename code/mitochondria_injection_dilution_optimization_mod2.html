           r_sigma_dilution*log(dilution_factor/0.5) +

           r_sigma_dilution_injection_interaction*log(injection_volume/5)*log(dilution_factor/0.5)) %>%
  mutate(ypred1 = r_metabolite +

           r_dilution*log(dilution_factor/0.5) +
           r_injection*log(injection_volume/5) +
