data_wide <- read.csv(here(data_dir, 'mito_dilop_stand_pos.csv')) %>% 
  mutate(IS = X237.081234046689.3.68837108833056, .after = time)

data_long_pos <- data_wide %>% 
  tidyr::gather(metabolite_code, intensity, 8:191, factor_key = TRUE) %>%
  rename(chromatogram_name = chrom,
         dilution_factor = dil,
         injection_volume = inj_vol,
         condition = cond,
         replicate = rep,
         chromatogram_number = time,
         internal_standard = IS) %>%
  mutate(metabolite_number =  match(metabolite_code, unique(metabolite_code))) %>%
  filter(intensity != 0) %>%
  mutate(dilution_factor_original = dilution_factor) %>%
  mutate(dilution_factor = if_else(metabolite_number == 184, 
                                   1, dilution_factor)) %>%
  mutate(metabolite_number = 1000 + metabolite_number) %>%
  mutate(log_intensity =  log(intensity),
         log_intensity_factor = log((intensity/(dilution_factor*injection_volume))),
         normalized_intensity = intensity/internal_standard,
         log_normalized_intensity = log(intensity/internal_standard),
         log_normalized_intensity_factor = log(normalized_intensity/(dilution_factor*injection_volume)),

  select(metabolite_number, dilution_factor, dilution_factor_original,
